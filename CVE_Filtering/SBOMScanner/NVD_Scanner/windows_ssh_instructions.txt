Install:
Add-WindowsCapability -Online -Name OpenSSH.Server~~~~0.0.1.0

Start and Enable:
Start-Service sshd
Set-Service -Name sshd -StartupType 'Automatic'

    See if enabled and stop if needed:
    Get-Service sshd 
    Stop-Service sshd

Allow SSh through firewall:
New-NetFirewallRule -Name sshd `
  -DisplayName 'OpenSSH Server (sshd)' `
  -Enabled True -Direction Inbound `
  -Protocol TCP -Action Allow -LocalPort 22

Create sshuser account:
net user sshuser YourStrongPass123! /add
net localgroup administrators sshuser /add

Get the IP address:
ipconfig

Connect:
ssh sshuser@ipaddress
